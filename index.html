function initMap(){

 map=L.map("alertMap").setView([39.29,-76.61],8);

 L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);

/* ===== ALERTS ===== */

let alertsData={};


function notifyUser(alert){

 if(alert.severity==="warning" && "Notification" in window){

  new Notification(`⚠️ ${alert.type}`, { body: alert.description });

 }

}


function showAlertOnMap(alert){

 fetch("https://nominatim.openstreetmap.org/search?format=json&q="+encodeURIComponent(alert.location))

 .then(res=>res.json())

 .then(data=>{

  if(!data.length) return;

  const lat=data[0].lat;

  const lon=data[0].lon;


  L.marker([lat,lon]).addTo(map)

   .bindPopup(`<b>${alert.type}</b><br>${alert.location}`);

 });

}


function renderAlerts(data){

 const list=document.getElementById("alertsList");

 list.innerHTML="";


 for(const key in data){

  const a=data[key];


  const div=document.createElement("div");

  div.className="alertCard alert-"+a.severity;


  div.innerHTML=`

  <b>${a.location}</b> - ${a.type}<br>

  ${a.description}

  <br><button onclick='showAlertOnMap(${JSON.stringify(a)})'>Map</button>

  `;


  list.appendChild(div);


  if(a.severity==="warning"){

   document.getElementById("warningBanner").style.display="block";

   notifyUser(a);

  }

 }

}


alertsRef.on("value",(snap)=>{

 alertsData=snap.val() || {};

 renderAlerts(alertsData);

});
